Import and Prepare Data

Code 
# Load data retrieval and processing function for DWR integrated discrete nutrient data
source(here("code/02_data_retrieval_processing/dwr_nutrients.R"))
nutr <- get_nutrients()

# Load data retrieval and processing function for integrated water quality data set from discretewq
source(here("code/02_data_retrieval_processing/discretewq.R"))
dwq <- get_discretewq()

# Count number of samples for each Year, Month, Region, and Parameter
ls_count_month <-
  list(
    nutr = nutr,
    dwq = dwq |>
      mutate(
        Parameter = replace_values(
          Parameter,
          "DissolvedOxygen" ~ "DO",
          "DissSilica" ~ "Dissolved Silica"
        )
      )
  ) |>
  map(\(x) count(x, Year, Month, Region, Parameter, name = "Num_samples"))

# Further summarize number of samples for each Year, Season, Region, and Parameter using
# Adjusted calendar year: December-November, with December of the previous calendar year included
# with the following year
ls_count_season <- ls_count_month |>
  map(
    \(x) {
      mutate(
        x,
        YearAdj = assign_year_adj(Month, Year),
        Season = assign_season(Month)
      ) |>
        summarize(
          Num_samples = sum(Num_samples),
          .by = c(YearAdj, Season, Region, Parameter)
        )
    }
  )

Sampling Effort Plots

Code 
# Monthly values
ls_count_month_plt <- ls_count_month |>
  map(
    \(x) {
      filter(x, Year %in% 2010:2024) |>
        mutate(Month = month(Month, label = TRUE)) |>
        complete(nesting(Year, Month), Region, Parameter) |>
        nest(.by = Parameter) |>
        deframe() |>
        imap(
          \(x, idx) {
            ggplot(data = x, aes(x = Year, y = Region, fill = Num_samples)) +
              geom_tile() +
              facet_wrap(vars(Month)) +
              ggtitle(idx) +
              scale_y_discrete(drop = FALSE) +
              scale_x_continuous(
                breaks = breaks_pretty(10),
                expand = expansion(mult = 0.02)
              ) +
              scale_fill_viridis_c(
                name = "Number of\nSamples",
                na.value = "transparent"
              ) +
              theme_bw() +
              theme(
                axis.text.x = element_text(angle = 90, hjust = 1, vjust = 0.5)
              )
          }
        )
    }
  )

# Purpose

Investigate sampling effort for model input data used in the DWR nutrient synthesis project. This page explores sampling effort for both the DWR integrated discrete nutrient data set and the integrated discrete water quality data from [discretewq](https://github.com/InteragencyEcologicalProgram/discretewq).
