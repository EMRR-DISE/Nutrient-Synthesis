Monthly Data

Prepare Data

Code 
df_monthly <- readRDS(here("data/processed/monthly_values.rds"))

df_monthly_c <- df_monthly |>
  mutate(
    Date = make_date(year = Year, month = Month),
    Region = factor(
      Region,
      levels = c(
        "North",
        "Confluence",
        "SouthCentral",
        "Suisun Marsh",
        "Suisun Bay"
      )
    )
  ) |>
  select(Date, Region, starts_with("bv_")) |>
  pivot_longer(
    cols = starts_with("bv_"),
    names_to = "Algal_Group",
    values_to = "Biovolume",
    values_drop_na = TRUE
  ) |>
  mutate(Algal_Group = str_remove(Algal_Group, "^bv_"))

Time-series Plots

Code 
plt_base_monthly <- df_monthly_c |>
  ggplot(aes(x = Date, y = Biovolume, fill = Algal_Group)) +
  scale_fill_muted(name = "Algal Group") +
  scale_x_date(
    name = "Date",
    breaks = breaks_pretty(10),
    expand = expansion(mult = 0.003)
  ) +
  theme_bw() +
  theme(legend.position = "top") +
  guides(fill = guide_legend(nrow = 1))

Stacked Bar Plots, fixed scale

Code 
plt_base_monthly +
  geom_col() +
  facet_wrap(vars(Region), ncol = 1)

Stacked Bar Plots, free scales

Code 
plt_base_monthly +

  facet_wrap(vars(Region), ncol = 1, scales = "free_y")

Proportional Bar Plots

Code 
plt_base_monthly +

  scale_y_continuous(
    name = "Proportion of Biovolume",
    labels = label_percent(),

Explore Algal Group trends for the DWR nutrient synthesis project.

library(tidyverse)

# Monthly Data

## Prepare Data

```{r}
df_monthly <- readRDS(here("data/processed/monthly_values.rds"))

df_monthly_c <- df_monthly |>
  mutate(
    Date = make_date(year = Year, month = Month),
    Region = factor(
      Region,
      levels = c(
        "North",
        "Confluence",
        "SouthCentral",
        "Suisun Marsh",
        "Suisun Bay"
      )
    )
  ) |>
  select(Date, Region, starts_with("bv_")) |>
  pivot_longer(
    cols = starts_with("bv_"),
    names_to = "Algal_Group",
    values_to = "Biovolume",
    values_drop_na = TRUE
  ) |>
  mutate(Algal_Group = str_remove(Algal_Group, "^bv_"))
```

## Time-series Plots

```{r}
plt_base_monthly <- df_monthly_c |>
  ggplot(aes(x = Date, y = Biovolume, fill = Algal_Group)) +
  scale_fill_muted(name = "Algal Group") +
  scale_x_date(
    name = "Date",
    breaks = breaks_pretty(10),
    expand = expansion(mult = 0.003)
  ) +
  theme_bw() +
  theme(legend.position = "top") +
  guides(fill = guide_legend(nrow = 1))
```

### Stacked Bar Plots, fixed scale

```{r}
#| fig-width: 9
#| fig-height: 9.5

plt_base_monthly +
  geom_col() +
  facet_wrap(vars(Region), ncol = 1)
```

### Stacked Bar Plots, free scales

```{r}
#| fig-width: 9
#| fig-height: 9.5

plt_base_monthly +
  geom_col() +
  facet_wrap(vars(Region), ncol = 1, scales = "free_y")
```

### Proportional Bar Plots

```{r}
#| fig-width: 9
#| fig-height: 9.5

plt_base_monthly +
  geom_col(position = "fill") +
  facet_wrap(vars(Region), ncol = 1) +
  scale_y_continuous(
    name = "Proportion of Biovolume",
    labels = label_percent(),
    expand = expansion(mult = 0.02)
  )
```
